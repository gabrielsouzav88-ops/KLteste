local function waitChain(root, names, timeout)
	local node = root
	for _, name in ipairs(names) do
		node = node:WaitForChild(name, timeout)
		if not node then
			return nil, name
		end
	end
	return node
end

local function aliveChar()
	local char = player.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if hum and hum.Health > 0 and char:FindFirstChild("HumanoidRootPart") then
		return char
	end
end

local function getRoot()
	local char = aliveChar()
	return char and char.HumanoidRootPart
end

local function getMobRoot(mob)
	return mob:FindFirstChild("HumanoidRootPart")
		or mob.PrimaryPart
		or mob:FindFirstChildWhichIsA("BasePart", true)
end

local function isAlive(mob)
	local hum = mob and mob:FindFirstChildOfClass("Humanoid")
	return hum and hum.Health > 0 and getMobRoot(mob)
end

local mobFolder
local warnedMobFolder = false

local function getMobFolder()
	if mobFolder and mobFolder.Parent then
		return mobFolder
	end
	local folder = workspace:FindFirstChild(MOB_FOLDER_NAME)
	if folder then
		mobFolder = folder
		return folder
	end
	if not warnedMobFolder then
		warnedMobFolder = true
		warn("[AutoFarm] workspace." .. MOB_FOLDER_NAME
			.. " nao encontrada: varrendo o workspace inteiro (mais pesado).")
	end
	return workspace
end

-- Filho por nome: tenta o nome exato e, se falhar, ignora maiusculas/minusculas
local function findChildCI(parent, name)
	local exact = parent:FindFirstChild(name)
	if exact then
		return exact
	end
	local lowered = name:lower()
	for _, child in ipairs(parent:GetChildren()) do
		if child.Name:lower() == lowered then
			return child
		end
	end
	return nil
end

-- Resolve um caminho de nomes a partir de workspace. Retorna instancia, ou nil e o nome que faltou.
local function resolveInWorkspace(parts)
	local node = workspace
	for _, name in ipairs(parts) do
		node = findChildCI(node, name)
		if not node then
			return nil, name
		end
	end
	return node
end

local LEVEL_NAMES = { level = true, lv = true, lvl = true, levels = true }
local cachedLevelObj
local lastLevelScan = -math.huge

local function getLevel()
	if LEVEL_PATH then
		local ok, v = pcall(LEVEL_PATH)
		if ok and v then
			return tonumber(typeof(v) == "Instance" and v.Value or v)
		end
	end

	if cachedLevelObj and cachedLevelObj.Parent then
		return tonumber(cachedLevelObj.Value)
	end

	for name, value in pairs(player:GetAttributes()) do
		if LEVEL_NAMES[name:lower()] and tonumber(value) then
			return tonumber(value)
		end
	end

	-- varredura de descendentes limitada a uma vez a cada 5 s
	if os.clock() - lastLevelScan < 5 then
		return nil
	end
	lastLevelScan = os.clock()

	for _, d in ipairs(player:GetDescendants()) do
		if (d:IsA("IntValue") or d:IsA("NumberValue"))
			and LEVEL_NAMES[d.Name:lower()] then
			cachedLevelObj = d
			return tonumber(d.Value)
		end
	end

	return nil
end

local function isTarget(obj, key)
	if not obj:IsA("Model") then
		return false
	end
	local hum = obj:FindFirstChildOfClass("Humanoid")
	if not hum then
		return false
	end
	if normalize(obj.Name):find(key, 1, true) == nil
		and normalize(hum.DisplayName):find(key, 1, true) == nil then
		return false
	end
	return not Players:GetPlayerFromCharacter(obj)
end

local estimatedLevels = {}
local lastScan = {}

-- Descobre o level pelo "[Lv. N]" do monstro. Retorna nil se nao achar.
local function estimateLevel(q)
	if estimatedLevels[q.Name] then
		return estimatedLevels[q.Name]
	end
	if os.clock() - (lastScan[q.Name] or -math.huge) < 5 then
		return nil
	end
	lastScan[q.Name] = os.clock()

	local lowest
	for _, obj in ipairs(getMobFolder():GetDescendants()) do
		if isTarget(obj, q.NpcKey) then
			local hum = obj:FindFirstChildOfClass("Humanoid")
			local text = obj.Name .. " " .. (hum and hum.DisplayName or "")
			local n = tonumber(text:match("[Ll][Vv]%.?%s*(%d+)"))
			if n and (not lowest or n < lowest) then
				lowest = n
			end
		end
	end

	if lowest then
		estimatedLevels[q.Name] = lowest
	end
	return lowest
end

local function requiredLevel(q)
	return q.MinLevel or estimateLevel(q)
end

-- Retorna: quest escolhida, proxima quest bloqueada, lista de quests sem level conhecido, level
local function pickQuest()
	if State.AutoByLevel then
		local level = getLevel()
		if level then
			local chosen, chosenReq = nil, -1
			local nextLocked
			local unknown = {}

			for _, q in ipairs(QUESTS) do
				local req = requiredLevel(q)
				if not req then
					table.insert(unknown, q)
				elseif level >= req then
					if req > chosenReq then
						chosen, chosenReq = q, req
					end
				elseif not nextLocked or req < nextLocked.req then
					nextLocked = { quest = q, req = req }
				end
			end

			return chosen or QUESTS[1], nextLocked, unknown, level
		end
	end
	return findQuestByName(State.ManualQuest), nil, {}, nil
end

-- InvokeServer em outra thread, com limite de tempo: se o servidor nao responder,
-- o farm nao congela. Retorna (ok, resultado); ok = false se deu erro ou estourou o tempo.
local function invokeWithTimeout(remote, args, timeout)
	local done, ok, result = false, false, nil
	task.spawn(function()
		ok, result = pcall(remote.InvokeServer, remote, table.unpack(args))
		done = true
	end)
	local startedAt = os.clock()
	while not done and State.Alive and os.clock() - startedAt < timeout do
		task.wait()
	end
	return done and ok, result
end

-- Teleporta ate o NPC de quest da ilha da quest. Retorna true se teleportou.
local function teleportToQuestNpc(q)
	local island = q.Island
	if not island then
		return false
	end
	local root = getRoot()
	if not root then
		return false
	end

	local npc, missing = resolveInWorkspace(island.Path)
	if not npc or not npc:IsA("PVInstance") then
		local why = npc and ("a instancia e " .. npc.ClassName .. ", nao uma parte ou modelo")
			or ("nao achei '" .. tostring(missing) .. "' no caminho")
		warnOnce("npc-" .. island.Name, "NPC da " .. island.Name .. ": " .. why
			.. ". Vou pedir a quest mesmo assim.")
		return false
	end

	root.CFrame = CFrame.new(npc:GetPivot().Position + NPC_TELEPORT_OFFSET)
	root.AssemblyLinearVelocity = Vector3.zero
	return true
end

-- Pega a quest. Se a quest tem ilha, teleporta ao NPC antes. Se nao tem Args, so teleporta.
local function takeQuest(q)
	if q.Island and (teleportedFor ~= q or (TELEPORT_EVERY_TAKE and q.Args ~= nil)) then
		if teleportToQuestNpc(q) then
			teleportedFor = q
			task.wait(NPC_SETTLE_TIME)
		end
	end

	if q.Args then
		invokeWithTimeout(QuestRemote, q.Args, QUEST_TIMEOUT)
	end
end

-- M1 sem bloquear o loop: no maximo uma chamada em andamento por vez
local m1Token = 0
local m1Pending = false
local m1StartedAt = 0

local function m1()
	if m1Pending and os.clock() - m1StartedAt < M1_STALE_AFTER then
		return
	end
	m1Token += 1
	local token = m1Token
	m1Pending = true
	m1StartedAt = os.clock()
	task.spawn(function()
		pcall(SkillRemote.InvokeServer, SkillRemote, table.unpack(M1Args))
		if token == m1Token then
			m1Pending = false
		end
	end)
end

----------------------------------------------------------------
-- Localizar mob via EtcFunction("GetEntityLocation", { EntityName = ... })
-- Eu NAO sei o que esse remote devolve: so uso o resultado se for Vector3, CFrame
-- ou uma instancia com posicao. Qualquer outra coisa e impressa para voce me mandar.

local lastLocateAt = -math.huge

local function describeValue(value)
	local text = typeof(value) .. ": " .. tostring(value)
	if type(value) == "table" then
		local parts, count = {}, 0
		for k, v in pairs(value) do
			count += 1
			if count > 20 then
				table.insert(parts, "...")
				break
			end
			table.insert(parts, tostring(k) .. "=" .. typeof(v) .. ":" .. tostring(v))
		end
		text = text .. " { " .. table.concat(parts, ", ") .. " }"
	end
	return text
end

local function locationToPosition(value)
	local kind = typeof(value)
	if kind == "Vector3" then
		return value
	elseif kind == "CFrame" then
		return value.Position
	elseif kind == "Instance" and value:IsA("PVInstance") then
		return value:GetPivot().Position
	end
	if value ~= nil then
		warnOnce("locate-type", "GetEntityLocation devolveu um tipo que eu nao sei usar. Veja o console e me mande.")
		print("[AutoFarm] GetEntityLocation ->", describeValue(value))
	end
	return nil
end

local function locateEntity(entityName)
	if not (EtcRemote and EtcRemote:IsA("RemoteFunction")) then
		warnOnce("etc-missing", "EtcFunction nao encontrado (ou nao e RemoteFunction): 'Locate' desativado.")
		return nil
	end
	if os.clock() - lastLocateAt < LOCATE_INTERVAL then
		return nil
	end
	lastLocateAt = os.clock()

	local ok, result = invokeWithTimeout(EtcRemote, { "GetEntityLocation", { EntityName = entityName } }, QUEST_TIMEOUT)
	if not ok then
		return nil
	end
	return locationToPosition(result)
end

local ActiveSkills = {}

for index, s in ipairs(SKILLS) do
	local problem
	if type(s.Name) ~= "string" or s.Name == "" then
		problem = "falta Name"
	elseif type(s.Args) ~= "table" and type(s.ArgsFn) ~= "function" then
		problem = "falta Args (tabela) ou ArgsFn (funcao)"
	elseif type(s.Cooldown) ~= "number" or s.Cooldown <= 0 then
		problem = "falta Cooldown (numero maior que 0)"
	end
	if problem then
		warn(("[AutoFarm] SKILLS[%d] ignorada: %s"):format(index, problem))
	else
		s.Category = s.Category or "Outras"
		s.On = s.Enabled == true
		s.NextUse = 0
		table.insert(ActiveSkills, s)
	end
end

local function skillRemoteFor(s)
	if s._remote and s._remote.Parent then
		return s._remote
	end
	local remote = Functions:FindFirstChild(s.Remote or "SkillAction")
	if remote and (remote:IsA("RemoteFunction") or remote:IsA("RemoteEvent")) then
		s._remote = remote
		return remote
	end
	return nil
end

-- Garante que a Tool esta equipada. Retorna (achou, equipouAgora)
local function ensureToolEquipped(toolName)
	local char = aliveChar()
	if not char then
		return false, false
	end
	local equipped = char:FindFirstChild(toolName)
	if equipped and equipped:IsA("Tool") then
		return true, false
	end
	local backpack = player:FindFirstChildOfClass("Backpack")
	local tool = backpack and backpack:FindFirstChild(toolName)
	local hum = char:FindFirstChildOfClass("Humanoid")
	if tool and tool:IsA("Tool") and hum then
		hum:EquipTool(tool)
		return true, true
	end
	return false, false
end

local function callRemote(remote, args)
	if remote:IsA("RemoteFunction") then
		task.spawn(function()
			pcall(remote.InvokeServer, remote, table.unpack(args))
		end)
		return true
	elseif remote:IsA("RemoteEvent") then
		pcall(remote.FireServer, remote, table.unpack(args))
		return true
	end
	return false
end

local function fireSkill(s, mob)
	local args = s.Args
	if s.ArgsFn then
		local ok, result = pcall(s.ArgsFn, mob)
		if not ok or type(result) ~= "table" then
			warnOnce("argsfn-" .. s.Name, "ArgsFn da skill '" .. s.Name .. "' falhou ou nao devolveu uma tabela.")
			return false
		end
		args = result
	end

	local remote = skillRemoteFor(s)
	if not remote then
		warnOnce("remote-" .. s.Name, "Remote '" .. (s.Remote or "SkillAction")
			.. "' da skill '" .. s.Name .. "' nao encontrado em Functions.")
		return false
	end

	if s.Tool then
		local found, changed = ensureToolEquipped(s.Tool)
		if not found then
			warnOnce("tool-" .. s.Name, "Tool '" .. s.Tool .. "' da skill '" .. s.Name
				.. "' nao esta no personagem nem no Backpack.")
			return false
		end
		if changed then
			task.wait(TOOL_EQUIP_WAIT)
		end
	end

	return callRemote(remote, args)
end

local lastSkillAt = 0

-- Usa no maximo UMA skill por chamada, respeitando cooldown e a pausa minima entre skills
local function useSkills(mob)
	if not State.UseSkills or #ActiveSkills == 0 then
		return
	end
	local now = os.clock()
	if now - lastSkillAt < State.SkillGap then
		return
	end
	for _, s in ipairs(ActiveSkills) do
		if s.On and now >= s.NextUse then
			s.NextUse = now + s.Cooldown -- mesmo se falhar, nao fica tentando a cada loop
			if fireSkill(s, mob) then
				lastSkillAt = os.clock()
				return
			end
		end
	end
end

-- mobs que estouraram o tempo ficam ignorados por um periodo
local skipped = setmetatable({}, { __mode = "k" })

local function findNearestMob(q)
	local root = getRoot()
	if not root then
		return nil
	end

	local now = os.clock()
	local best, bestDist
	for _, obj in ipairs(getMobFolder():GetDescendants()) do
		if isTarget(obj, q.NpcKey) and isAlive(obj) and (skipped[obj] or 0) < now then
			local dist = (getMobRoot(obj).Position - root.Position).Magnitude
			if not bestDist or dist < bestDist then
				best, bestDist = obj, dist
			end
		end
	end

	return best
end

-- Ataca ate o mob morrer, ou ate VOCE morrer / desligar o farm / estourar o tempo
local function killMob(mob)
	local hum = mob:FindFirstChildOfClass("Humanoid")
	local startedAt = os.clock()

	while State.Alive and State.Farming and isAlive(mob) do
		local root = getRoot()
		if not root then
			return false -- voce morreu
		end

		if os.clock() - startedAt > MOB_TIMEOUT then
			skipped[mob] = os.clock() + MOB_SKIP_TIME
			return false
		end

		root.CFrame = getMobRoot(mob).CFrame * CFrame.new(0, 0, State.Distance)
		root.AssemblyLinearVelocity = Vector3.zero
		m1()
		useSkills(mob)
		task.wait(State.AttackDelay)
	end

	if hum and hum.Health <= 0 then
		State.Kills += 1
		return true
	end
	return false
end

local function getStatsRemote()
	local ok, remote = pcall(function()
		return player.PlayerGui.MainGui.StarterFrame.StatsFrame.RemoteEvent
	end)
	if ok and remote and remote:IsA("RemoteEvent") then
		return remote
	end
	return nil
end

local function upgradeStat(stat, amount)
	local remote = getStatsRemote()
	if not remote then
		return false
	end
	pcall(function()
		remote:FireServer(stat, amount)
	end)
	return true
end

local function upgradeSelected()
	local any = false
	for _, stat in ipairs(STAT_NAMES) do
		if State.Stats[stat] then
			any = true
			if not upgradeStat(stat, State.StatAmount) then
				return false, "Remote dos status nao encontrado (ou nao e um RemoteEvent)."
			end
		end
	end
	if not any then
		return false, "Selecione ao menos um status."
	end
	return true
end

local function debugScan()
	local lines = 0
	local function out(...)
		if lines < 250 then
			print(...)
		end
		lines += 1
	end

	out("---- MONSTROS (nomes unicos) ----")
	local seen = {}
	for _, obj in ipairs(getMobFolder():GetDescendants()) do
		if obj:IsA("Model") and obj:FindFirstChildOfClass("Humanoid")
			and not Players:GetPlayerFromCharacter(obj) and not seen[obj.Name] then
			seen[obj.Name] = true
			out(obj:GetFullName())
		end
	end

	out("---- NPCs DE QUEST DAS ILHAS ----")
	for _, island in ipairs(ISLANDS) do
		local npc, missing = resolveInWorkspace(island.Path)
		if npc then
			local pos = npc:IsA("PVInstance") and tostring(npc:GetPivot().Position) or "-"
			out("OK", island.Name, npc.ClassName, npc:GetFullName(), pos)
		else
			out("NAO ENCONTRADO", island.Name, "faltou o nome:", tostring(missing))
		end
	end

	out("---- QUEST -> ILHA ----")
	for _, q in ipairs(QUESTS) do
		out(q.Name, "| level", tostring(q.MinLevel), "| ilha", q.Island and q.Island.Name or "-")
	end

	out("---- SKILLS ATIVAS NA CONFIG ----")
	out("quantidade:", #ActiveSkills, "| EtcFunction:", EtcRemote and EtcRemote.ClassName or "nao encontrado")

	out("---- POSSIVEIS DADOS DE QUEST ----")
	for _, root in ipairs({ ReplicatedStorage, workspace }) do
		for _, obj in ipairs(root:GetDescendants()) do
			if obj.Name:lower():find("quest") then
				out(obj.ClassName, obj:GetFullName())
			end
		end
	end

	if lines > 250 then
		print(("(... %d linhas omitidas)"):format(lines - 250))
	end

	notify("Scan", "Resultado impresso no console do executor.", 5)
end

-- Loop do auto-upar status
----------------------------------------------------------------
task.spawn(function()
	local warned = false
	local nextRun = 0
	while State.Alive do
		task.wait(0.5)
		if State.AutoStats and os.clock() >= nextRun then
			local ok, err = upgradeSelected()
			if not ok and not warned then
				warned = true
				notify("Status", err, 5)
			elseif ok then
				warned = false
			end
			nextRun = os.clock() + State.StatInterval
		end
	end
end)

----------------------------------------------------------------
-- Loop principal do farm
----------------------------------------------------------------
task.spawn(function()
	while State.Alive do
		task.wait()
		if State.Farming then
			-- Morreu? espera o respawn e recomeca
			if not aliveChar() then
				if player.Character then
					State.Deaths += 1 -- so conta se havia personagem (e nao so carregando)
				end
				updateStatus("Morreu, aguardando respawn...")
				repeat
					task.wait(0.5)
				until aliveChar() or not State.Farming or not State.Alive

				if State.Farming and State.Alive then
					task.wait(2) -- deixa o personagem carregar
					resetQuestState() -- repega a quest (e vai ao NPC da ilha de novo)
				end
			end

			if State.Farming and State.Alive then
				local q, nextLocked, unknown, level = pickQuest()

				-- Avisos sobre quests bloqueadas / sem level conhecido
				local info = ""
				if nextLocked then
					info = string.format(
						"Proxima quest: %s (level %d). Voce esta no level %s, continuando o farm.",
						nextLocked.quest.Name, nextLocked.req, tostring(level)
					)
				end
				if #unknown > 0 then
					local list = {}
					for _, u in ipairs(unknown) do
						table.insert(list, u.Name)
					end
					info = info .. (info ~= "" and "\n" or "")
						.. "Nao consegui descobrir o level de: " .. table.concat(list, ", ")
						.. ". Defina MinLevel no script."
				end
				if State.AutoByLevel and not level then
					info = "Level do jogador nao encontrado, usando a quest manual."
				end
				if info ~= "" and info ~= State.LastInfo then
					State.LastInfo = info
					notify("Quests", info, 8)
				end

				if q ~= currentQuest then
					currentQuest = q
					takeQuest(q)
					notify("Quest", q.Name .. (q.Island and (" | " .. q.Island.Name) or ""), 4)
				end

				local mob = findNearestMob(q)

				if mob then
					updateStatus("Farmando: " .. q.Npc)
					local killed = killMob(mob)
					if killed and State.Farming and State.AutoQuest and q.Args then
						takeQuest(q)
					end
				else
					updateStatus("Procurando '" .. q.Npc .. "'...")
					if q.Locate then
						local root = getRoot()
						local pos = locateEntity(q.Locate)
						if root and pos then
							root.CFrame = CFrame.new(pos + LOCATE_HOVER)
							root.AssemblyLinearVelocity = Vector3.zero
						end
					end
					task.wait(0.5)
				end
			end
		end
	end
end)
